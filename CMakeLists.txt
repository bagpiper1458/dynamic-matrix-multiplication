cmake_minimum_required(VERSION 3.16)
project(dynamic_matrix LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(dm_test cpp/test.cpp)
add_executable(dm_benchmark cpp/benchmark.cpp)
add_executable(dm_compare cpp/compare.cpp)
add_executable(dm_compare_packed cpp/compare_packed.cpp)
foreach(target dm_test dm_benchmark dm_compare dm_compare_packed)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
enable_testing()
add_test(NAME cpp_correctness COMMAND dm_test "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures.txt")
